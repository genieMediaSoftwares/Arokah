# ─────────────────────────────────────────────────────────────────────────────
# Sample public_html/.htaccess
#
# Rename to ".htaccess" in public_html. If you already have one for the React
# build, do not replace it wholesale — merge the four blocks below into it.
#
# Every block here exists because of a failure that presents as a CORS error in
# the browser while having nothing to do with CORS. Read the comments before
# deleting any of them.
# ─────────────────────────────────────────────────────────────────────────────


# ─────────────────────────────────────────────────────────────────────────────
# 1. Restore the Authorization header
#
# On CGI and FastCGI — which is how Hostinger runs PHP — Apache does not pass
# Authorization through to the script, because a CGI process could otherwise
# read credentials meant for the server itself. PHP then sees no bearer token
# and every upload returns 401 "Sign in again", no matter how valid the token is.
#
# Note the RewriteCond. %1 is a backreference to the LAST RewriteCond that ran;
# with no preceding condition it expands to an empty string and the rule silently
# does nothing. The condition is not optional decoration — it is where the value
# comes from.
# ─────────────────────────────────────────────────────────────────────────────
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteCond %{HTTP:Authorization} ^(.+)$
    RewriteRule .* - [E=HTTP_AUTHORIZATION:%1]
</IfModule>

# Same goal, different mechanism, for servers where the rewrite above is not
# enough. Harmless when it is redundant — _upload_lib.php reads every spelling.
<IfModule mod_setenvif.c>
    SetEnvIf Authorization "(.*)" HTTP_AUTHORIZATION=$1
</IfModule>


# ─────────────────────────────────────────────────────────────────────────────
# 2. The SPA rewrite — THE most common cause of "preflight failed"
#
# React Router needs unknown URLs sent to index.html. Written without the two
# conditions below, that rule also captures /upload.php: the browser's OPTIONS
# preflight is answered with your homepage's HTML, which carries no CORS headers,
# and the console reports
#
#     Response to preflight request doesn't pass access control check
#
# even though upload.php is perfectly fine and was never executed.
#
# `!-f` and `!-d` mean "only rewrite when no real file or directory matches",
# which is what keeps upload.php, delete.php, list.php and /uploads/* reachable.
# ─────────────────────────────────────────────────────────────────────────────
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # Belt and braces: never rewrite the upload API, even if the conditions
    # below are later edited or reordered by mistake.
    RewriteRule ^(upload|delete|list)\.php$ - [L]

    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule ^ index.html [L]
</IfModule>


# ─────────────────────────────────────────────────────────────────────────────
# 3. CORS fallback for responses PHP never produced
#
# _upload_lib.php applies CORS to every response it controls, including errors —
# that is the primary implementation and this does not replace it. This block
# covers only the cases where PHP never ran at all and therefore could not:
# a parse error in the PHP itself, a 404 for a file that was not uploaded, a
# 500 from the server before the interpreter started.
#
# Without it those failures reach the browser as a bare CORS error with the real
# cause invisible. With it, the browser can read the response and DevTools shows
# the actual status.
#
# `set` (not `add`) so this can only ever REPLACE the identical value PHP already
# sent — two Access-Control-Allow-Origin headers on one response is itself a CORS
# failure. `merge` on Vary for the same reason.
#
# Keep this regex in step with UPLOAD_DEFAULT_ORIGINS in _upload_lib.php.
# ─────────────────────────────────────────────────────────────────────────────
<IfModule mod_setenvif.c>
    SetEnvIf Origin "^https?://(localhost:5173|localhost:3000|(www\.)?arokah\.kkdigitalgrowth\.com|maroon-pig-939052\.hostingersite\.com)$" CORS_ORIGIN=$0
</IfModule>

<IfModule mod_headers.c>
    Header always set Access-Control-Allow-Origin %{CORS_ORIGIN}e env=CORS_ORIGIN
    Header always set Access-Control-Allow-Credentials "true" env=CORS_ORIGIN
    Header always set Access-Control-Allow-Methods "GET, POST, PUT, PATCH, DELETE, OPTIONS" env=CORS_ORIGIN
    Header always set Access-Control-Allow-Headers "Authorization, Content-Type, Accept, X-Requested-With" env=CORS_ORIGIN
    Header always set Access-Control-Max-Age "86400" env=CORS_ORIGIN
    Header always merge Vary Origin
</IfModule>

# Note this does NOT rescue the case in block 2. If the rewrite swallows
# upload.php, the preflight passes but the POST then receives index.html instead
# of JSON, and axios fails parsing it. Block 2 is the fix; this only makes the
# failure legible.


# ─────────────────────────────────────────────────────────────────────────────
# 4. Protect the config, and cache the build
#
# The config file holds the secret shared with the Express API. PHP is executed
# rather than served, so its contents are never exposed — but if PHP is ever
# disabled or misconfigured, this is the difference between a broken page and a
# leaked secret.
# ─────────────────────────────────────────────────────────────────────────────
<FilesMatch "^_upload_(config|config\.example|lib)\.php$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

# Do not compress images; they are already compressed and gzip only adds CPU.
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json
</IfModule>

<IfModule mod_headers.c>
    # The hashed asset filenames Vite emits are safe to cache forever.
    <FilesMatch "\.(?i:js|css|woff2)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>
    # index.html must NOT be cached, or a deploy leaves visitors on the old build.
    <FilesMatch "^index\.html$">
        Header set Cache-Control "no-cache, must-revalidate"
    </FilesMatch>
</IfModule>
